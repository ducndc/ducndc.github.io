422 sManagedRadio &get_radio_context(const std::string &fronthaul_iface)
423 {
424 auto zwdfs = m_zwdfs_radio_manager && m_zwdfs_radio_manager->first == fronthaul_iface;
425
426 auto it = m_radio_managers.find(fronthaul_iface);
427 if (!zwdfs && it == m_radio_managers.end()) {
428 LOG(DEBUG) << "Added new interface to radio managers: " << fronthaul_iface;
429 // Insert empty new element
430 it = m_radio_managers.emplace(fronthaul_iface, sManagedRadio{}).first;
431 }
432
433 return zwdfs ? m_zwdfs_radio_manager->second : it->second;
434 }
435
436 public:
437 sManagedRadio &operator[](const std::string &fronthaul_iface)
438 {
439 return get_radio_context(fronthaul_iface);
440 }
441
442 std::string get_radio_iface_from_fd(int fd)
443 {
444 if (fd == net::FileDescriptor::invalid_descriptor) {
445 return {};
446 }
447 for (auto &radio : get()) {
448 auto &radio_manager = radio.second;
449 if (fd == radio_manager.ap_manager_fd || fd == radio_manager.monitor_fd) {
450 return radio.first; // interface name
451 }
452 }
453 const auto &zwdfs_radio = get_zwdfs();
454 if (!zwdfs_radio) {
455 return {};
456 }
457 const auto &radio_manager = zwdfs_radio->second;
458 if (fd == radio_manager.ap_manager_fd || fd == radio_manager.monitor_fd) {
459 return zwdfs_radio->first; // interface name
460 }
461 return {};
462 }
463
464 std::map<std::string, sManagedRadio> &get() { return m_radio_managers; }

466 {
467 return m_zwdfs_radio_manager;
468 }

470 void set_zwdfs(std::map<std::string, sManagedRadio>::iterator &it)
471 {
472 m_zwdfs_radio_manager = std::make_unique<std::pair<std::string, sManagedRadio>>();
473 *m_zwdfs_radio_manager = std::move(*it);
474 m_radio_managers.erase(it);

484 bool do_on_each_radio_manager(
485 std::function<bool(sManagedRadio &radio_manager, const std::string &fronthaul_iface)>
486 operation)
487 {
488 auto success = true;
489 for (auto &radio : get()) {
490 success &= operation(radio.second, radio.first);
491 }
492 auto &zwdfs_radio = get_zwdfs();
493 if (!zwdfs_radio) {
494 return success;
495 }
496 success &= operation(zwdfs_radio->second, zwdfs_radio->first);
497 return success;
498 };
