97 bool dwpal_get_scan_params_fg(sScanCfgParams &params, size_t &result_size)
98 {
99 ssize_t received_result_size =
100 dwpal_nl_cmd_get(m_radio_info.iface_name, LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS,
101 m_nl_buffer, NL_MAX_REPLY_BUFFSIZE);
102 if (received_result_size <= 0) {
103 LOG(ERROR) << "LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS failed!"
104 << " received size <= 0";
105 result_size = ScanCfgParams_size_invalid;
106 return false;
107 }
108 result_size = received_result_size;
109 if (result_size != ScanCfgParams_size) {
110 LOG(ERROR) << "LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS failed!"
111 << " expected size = " << ScanCfgParams_size
112 << ", received size = " << result_size;
113 return false;
114 }
115
116 std::copy_n(m_nl_buffer, result_size, reinterpret_cast<unsigned char *>(&params));
117 return true;
118 }
119
120 bool dwpal_get_scan_params_bg(sScanCfgParamsBG &params, size_t &result_size)
121 {
122 ssize_t received_result_size =
123 dwpal_nl_cmd_get(m_radio_info.iface_name, LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS_BG,
124 m_nl_buffer, NL_MAX_REPLY_BUFFSIZE);
125 if (received_result_size <= 0) {
126 LOG(ERROR) << "LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS_BG failed!"
127 << " received size <= 0";
128 result_size = ScanCfgParams_size_invalid;
129 return false;
130 }
131 result_size = received_result_size;
132 if (result_size < ScanCfgParamsBG_min_size) {
133 LOG(ERROR) << "LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS_BG failed!"
134 << " expected minimal size = " << ScanCfgParamsBG_min_size
135 << ", received size = " << result_size;
136 return false;
137 }
138 if (result_size > ScanCfgParamsBG_size) {
139 LOG(ERROR) << "LTQ_NL80211_VENDOR_SUBCMD_GET_SCAN_PARAMS_BG failed!"
140 << " expected maximal size = " << ScanCfgParamsBG_size
141 << ", received size = " << result_size;
142 return false;
143 }
144
145 std::copy_n(m_nl_buffer, result_size, reinterpret_cast<unsigned char *>(&params));
146 return true;
147 }
