726 static std::string get_binary_association_frame(const char assoc_req[])
727 {
728 auto sub_str = std::string(assoc_req).substr(0, ASSOCIATION_FRAME_SIZE - 1);
729 if (sub_str.length() <= frame_body_idx) {
730 return {};
731 }
732 sub_str.erase(0, frame_body_idx);
733
734 //convert the hex string to binary
735 return beerocks::string_utils::hex_to_bytes<std::string>(sub_str);
