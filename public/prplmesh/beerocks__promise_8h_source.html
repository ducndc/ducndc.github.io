19template <class T> class promise {
20public:
21 promise() : m_value(T()), m_signal(false){};

24 // Set value for waiting threads
25 void set_value(const T value)
26 {
27 int err = pthread_mutex_lock(&m_mut);
28 mapf_assert(err == 0);
29 m_value = value;
30 m_signal = 1;
31 err = pthread_cond_broadcast(&m_cond);
32 mapf_assert(err == 0);
33 err = pthread_mutex_unlock(&m_mut);
34 mapf_assert(err == 0);
35 }
36 // Wait for a value to be set
37 void wait()
38 {
39 // lock the mutex and wait for the conditional variable to be set
40 int err = pthread_mutex_lock(&m_mut);
41 mapf_assert(err == 0);
42 // If signaled before the wait, no need to wait
43 if (m_signal == 1) {
44 m_signal = 0;
45 } else {
46 err = pthread_cond_wait(&m_cond, &m_mut);
47 mapf_assert(err == 0);
48 }
49 err = pthread_mutex_unlock(&m_mut);
50 mapf_assert(err == 0);
51 }
52
53 // Wait for a value to be set with timeout
54 bool wait_for(uint32_t timeout_ms)
55 {
56 struct timeval now;
57 struct timespec timeout;
58 // set the absolute timeout
59 gettimeofday(&now, NULL);
60 timeout.tv_sec = now.tv_sec + (timeout_ms / 1000);
61 timeout.tv_nsec = (now.tv_usec + (timeout_ms % 1000) * 1000) * 1000;
62
63 // lock the mutex and wait for the conditional variable to be set or timeout
64 int err = pthread_mutex_lock(&m_mut);
65 int err_wait = 0;
66 mapf_assert(err == 0);
67 // If signaled before the wait, no need to wait
68 if (m_signal == 1) {
69 m_signal = 0;
70 } else {
71 if ((err_wait = pthread_cond_timedwait(&m_cond, &m_mut, &timeout)) != 0) {
72 LOG(ERROR) << "pthread_cond_timedwait failed, error code: " << err_wait;
73 }
74 }
75 err = pthread_mutex_unlock(&m_mut);
76 mapf_assert(err == 0);
77 if (err_wait)
78 return false;
79 return true;
80 }
81
82 T get_value() { return m_value; }
83
84private:
85 pthread_cond_t m_cond = PTHREAD_COND_INITIALIZER;
86 pthread_mutex_t m_mut = PTHREAD_MUTEX_INITIALIZER;
87 T m_value;
88 bool m_signal;
89};
