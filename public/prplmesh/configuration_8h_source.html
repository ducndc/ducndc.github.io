300template <class func, class ap_predicate>
301void Configuration::for_all_ap_vaps(func f, ap_predicate pred)
302{
303 auto f_with_iter = [&f](const std::string &vap, int iter) { f(vap); };

305 int dummy(0);
306 for_all_ap_vaps(f_with_iter, dummy, 10, pred);

309template <class func, class iter, class ap_predicate>
310void Configuration::for_all_ap_vaps(func f, iter current_iter, const iter end, ap_predicate pred)
311{
312 for_each(m_hostapd_config_vaps.begin(), m_hostapd_config_vaps.end(),
313 [this, &f, &current_iter, &end,
314 &pred](const std::pair<std::string, std::vector<std::string>> &vap) {
315 if (pred(vap.first)) {
316 if (end == current_iter) {
317 f(vap.first, end);
318 } else {
319 f(vap.first, current_iter++);
320 }
321 }
322 });
323}
